cmake_minimum_required(VERSION 3.20)
project(hao_x86_disasm LANGUAGES C CXX ASM)
add_subdirectory("${LLVM_SOURCE}/llvm" llvm EXCLUDE_FROM_ALL)
add_library(hao_x86_disasm SHARED initialize.cpp)
target_include_directories(hao_x86_disasm PRIVATE "${LLVM_SOURCE}/llvm/include" "${CMAKE_CURRENT_BINARY_DIR}/llvm/include")
llvm_map_components_to_libnames(disasm_libs X86Disassembler X86Desc X86Info MCDisassembler Support)
target_link_libraries(hao_x86_disasm PRIVATE ${disasm_libs})
target_compile_features(hao_x86_disasm PRIVATE cxx_std_17)
target_link_options(hao_x86_disasm PRIVATE "-Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.txt" -Wl,-dead_strip)
set_target_properties(hao_x86_disasm PROPERTIES OUTPUT_NAME hao-x86-disasm INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR YES)
